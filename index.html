Pilotaggio diretto PWM dei 3 canali di colore (Rosso, Verde, Blu)
Lettura dinamica del pulsante BOOT (GPIO0) per cambio colore
NexusCore_PWM_RGB.ino 
// Mappatura Pin PWM per LED RGB Analogico 4-Pin (con resistenze R1, R3, R5)
#define LED_R_PIN   4    // GPIO Canale Rosso (attraverso R1)
#define LED_G_PIN   5    // GPIO Canale Verde (attraverso R3)
#define LED_B_PIN   6    // GPIO Canale Blu (attraverso R5)
#define BOOT_BTN    0    // GPIO Pulsante BOOT Tattile

// Impostare a true se il LED è a Anodo Comune, false se a Catodo Comune
const bool COMMON_ANODE = false;

void setColor(uint8_t r, uint8_t g, uint8_t b) {
  if (COMMON_ANODE) {
    r = 255 - r;
    g = 255 - g;
    b = 255 - b;
  }
  analogWrite(LED_R_PIN, r);
  analogWrite(LED_G_PIN, g);
  analogWrite(LED_B_PIN, b);
}

void setup() {
  Serial.begin(115200);
  pinMode(BOOT_BTN, INPUT_PULLUP);
  
  pinMode(LED_R_PIN, OUTPUT);
  pinMode(LED_G_PIN, OUTPUT);
  pinMode(LED_B_PIN, OUTPUT);
  
  // Test Iniziale LED RGB (Sequenza R-G-B)
  setColor(255, 0, 0); delay(300);
  setColor(0, 255, 0); delay(300);
  setColor(0, 0, 255); delay(300);
}

void loop() {
  int state = digitalRead(BOOT_BTN);
  
  if (state == LOW) {
    // Pulsante Premuto -> Viola Nexus (R:255, G:0, B:200)
    setColor(255, 0, 200);
  } else {
    // Stato Normale -> Verde SMT (R:0, G:255, B:120)
    setColor(0, 255, 120);
  }
  delay(50);
}
